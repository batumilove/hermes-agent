_READ_CONNECTION_STACK: ContextVar[ExitStack | None] = ContextVar(
    "hermes_insights_read_connection_stack", default=None
)


def closes_read_connections(func):
    @wraps(func)
    def wrapped(*args, **kwargs):
        with ExitStack() as stack:
            token = _READ_CONNECTION_STACK.set(stack)
            try:
                return func(*args, **kwargs)
            finally:
                _READ_CONNECTION_STACK.reset(token)

    return wrapped


def open_read_connection(db: Path) -> sqlite3.Connection:
    stack = _READ_CONNECTION_STACK.get()
    if stack is None:
        raise RuntimeError("SQLite read connection opened outside a managed query scope")
    con = sqlite3.connect(f"file:{quote(str(db.resolve()), safe='/')}?mode=ro", uri=True)
    stack.callback(con.close)
    con.row_factory = sqlite3.Row
    return con


@closes_read_connections
def summary_readiness(db: Path, now_ts: float | None = None, max_age_seconds: float = 1800) -> dict[str, Any]:
    """Return producer-owned freshness state; request time never advances it."""
    con = open_read_connection(db)
    meta = {str(r["key"]): str(r["value"]) for r in con.execute("SELECT key,value FROM summary_meta")}
    generated = float(meta.get("summary_generated_at", "0") or 0)
    observed = float(meta.get("source_observed_at", "0") or 0)
    source_high = int(meta.get("source_message_high_water", "0") or 0)
    summary_high = int(meta.get("committed_message_high_water", "0") or 0)
    bootstrap = meta.get("bootstrap_complete") == "1"
    age = max(0.0, float(now_ts if now_ts is not None else time.time()) - generated) if generated else float("inf")
    caught_up = summary_high >= source_high
    latest_status = "unknown"
    latest_failed = False
    receipt = Path(str(db) + ".receipt.json")
    try:
        receipt_data = json.loads(receipt.read_text(encoding="utf-8"))
        latest_status = str(receipt_data.get("status", "unknown"))
        receipt_finished = float(receipt_data.get("finished_at", 0) or 0)
        latest_failed = latest_status in {"failed", "schema_blocked", "budget_blocked"} and receipt_finished > generated
    except (OSError, ValueError, TypeError, json.JSONDecodeError):
        pass
    ready = bool(generated and bootstrap and caught_up and age <= max_age_seconds and not latest_failed)
    if latest_failed:
        status = "generation_failed"
    elif not generated or not bootstrap:
        status = "incomplete"
    elif not caught_up:
        status = "lagging"
    elif age > max_age_seconds:
        status = "stale"
    else:
        status = "fresh"
    return {
        "ready": ready,
        "data_status": status,
        "summary_generated_at": generated,
        "source_observed_at": observed,
        "source_message_high_water": source_high,
        "summary_message_high_water": summary_high,
        "bootstrap_complete": bootstrap,
        "freshness_seconds": age,
        "max_age_seconds": max_age_seconds,
        "latest_generation_status": latest_status,
    }


def dt(ts: float | None) -> datetime | None:
    if ts is None:
        return None
    try:
        return datetime.fromtimestamp(float(ts), tz=timezone.utc).astimezone()
    except Exception:
        return None


def iso(ts: float | None) -> str:
    d = dt(ts)
    return d.isoformat(timespec="seconds") if d else ""


def day_key(ts: float | None) -> str:
    d = dt(ts)
    return d.strftime("%Y-%m-%d") if d else "unknown"


def hour_key(ts: float | None) -> int:
    d = dt(ts)
    return d.hour if d else 0


def weekday_key(ts: float | None) -> str:
    d = dt(ts)
    return d.strftime("%a") if d else "unknown"


def compact_num(n: float | int | None) -> str:
    if n is None:
        return "0"
    n = float(n)
    sign = "-" if n < 0 else ""
    n = abs(n)
    if n >= 1_000_000_000:
        return f"{sign}{n/1_000_000_000:.2f}B"
    if n >= 1_000_000:
        return f"{sign}{n/1_000_000:.2f}M"
    if n >= 1_000:
        return f"{sign}{n/1_000:.1f}K"
    if n == int(n):
        return f"{sign}{int(n)}"
    return f"{sign}{n:.2f}"


def fmt_int(n: float | int | None) -> str:
    return f"{int(n or 0):,}"


def fmt_money(n: float | None) -> str:
    return f"${(n or 0):,.2f}"


def duration(seconds: float | None) -> str:
    s = max(0, int(seconds or 0))
    if s >= 86400:
        return f"{s/86400:.1f}d"
    if s >= 3600:
        return f"{s/3600:.1f}h"
    if s >= 60:
        return f"{s//60}m"
    return f"{s}s"


def safe_model(v: str | None) -> str:
    return v or "unknown"


CRON_OVERDUE_GRACE_SECONDS = 3600
CRON_FALLBACK_STALE_HOURS = 48


def cron_schedule_state(job: dict[str, Any], now_ts: float | None = None) -> dict[str, Any]:
    """Classify a cron job from metadata without exposing its prompt or output."""
    now_ts = float(now_ts or time.time())
    def parse_ts(value: str | None) -> float | None:
        if not value:
            return None
        try:
            return datetime.fromisoformat(value.replace("Z", "+00:00")).timestamp()
        except Exception:
            return None

    last_ts = parse_ts(job.get("last_run_at"))
    next_ts = parse_ts(job.get("next_run_at"))
    enabled = bool(job.get("enabled", False))
    paused = job.get("state") == "paused" or bool(job.get("paused_at"))
    dormant = not enabled or paused
    run_age_hours = None if last_ts is None else max(0.0, (now_ts - last_ts) / 3600)
    overdue = False
    overdue_hours = 0.0
    schedule_basis = "next_run_at"
    if enabled and not paused and last_ts is not None:
        if next_ts is not None:
            overdue_hours = max(0.0, (now_ts - next_ts - CRON_OVERDUE_GRACE_SECONDS) / 3600)
            overdue = overdue_hours > 0
        else:
            schedule_basis = "48h_fallback"
            overdue = bool(run_age_hours is not None and run_age_hours > CRON_FALLBACK_STALE_HOURS)
            overdue_hours = max(0.0, float(run_age_hours or 0) - CRON_FALLBACK_STALE_HOURS)
    never_run = enabled and not paused and last_ts is None
    delivery = bool(job.get("last_delivery_error"))
    failed = bool(job.get("last_status") and job.get("last_status") != "ok")
    if dormant:
        classification = "dormant"
    elif failed:
        classification = "failed"
    elif delivery:
        classification = "delivery"
    elif never_run:
        classification = "never_run"
    elif overdue:
        classification = "overdue"
    else:
        classification = "healthy"
    return {
        "classification": classification,
        "execution_status": "failed" if failed else ("ok" if job.get("last_status") == "ok" else "unknown"),
        "delivery_status": "failed" if delivery else "ok",
        "overdue": overdue,
        "overdue_hours": overdue_hours,
        "never_run": never_run,
        "dormant": dormant,
        "run_age_hours": run_age_hours,
        "next_run_valid": next_ts is not None,
        "schedule_basis": schedule_basis,
    }


def total_tokens(row: dict[str, Any]) -> int:
    return int(row.get("input_tokens") or 0) + int(row.get("output_tokens") or 0) + int(row.get("cache_read_tokens") or 0) + int(row.get("cache_write_tokens") or 0) + int(row.get("reasoning_tokens") or 0)


def query_rows(con: sqlite3.Connection, sql: str, params: tuple = ()) -> list[dict[str, Any]]:
    cur = con.execute(sql, params)
    cols = [d[0] for d in cur.description]
    return [dict(zip(cols, r)) for r in cur.fetchall()]






def build_archive_trends(history: list[dict[str, Any]]) -> list[dict[str, Any]]:
    rows = list(reversed(history[-90:] if len(history) > 90 else history))
    specs = [
        ("sessions", "Sessions/day"),
        ("messages", "Messages/day"),
        ("total_tokens", "Tokens/day"),
        ("cost_usd", "Cost/day"),
        ("anomalies", "Anomalies/day"),
        ("cron_overdue", "Overdue cron/day"),
    ]
    out = []
    for key, label in specs:
        pts = [{"date": r.get("date"), "value": float(r.get(key) or 0)} for r in rows]
        vals = [p["value"] for p in pts]
        out.append({"key": key, "label": label, "points": pts, "max": max(vals) if vals else 0, "latest": vals[-1] if vals else 0})
    return out

def build_attention(data: dict[str, Any]) -> dict[str, Any]:
    cron = ((data.get("cron_health") or {}).get("summary") or {})
    anomalies = data.get("anomalies") or []
    active = data.get("active_sessions") or []
    cache = data.get("cache") or {}
    archive = (data.get("archive_history") or [{}])[0] if data.get("archive_history") else {}
    items: list[dict[str, Any]] = []
    def add(level: str, label: str, value: int | float | str, href: str = "") -> None:
        items.append({"level": level, "label": label, "value": value, "href": href})
    delivery_errors = int(cron.get("delivery_errors") or 0)
    status_issues = int(cron.get("last_status_not_ok") or 0)
    overdue_cron = int(cron.get("overdue") or cron.get("stale_enabled_48h") or 0)
    anomaly_count = len(anomalies)
    active_count = len(active)
    cache_age = int(cache.get("age_seconds") or 0)
    if delivery_errors:
        add("red", "cron delivery errors", delivery_errors, "#cron-delivery-errors")
    if status_issues:
        add("red", "cron status issues", status_issues, "#cron-status-issues")
    if overdue_cron:
        add("amber", "overdue cron jobs", overdue_cron, "#overdue-cron")
    if anomaly_count:
        add("amber", "cost/token anomalies", anomaly_count, "#anomalies")
    if active_count:
        add("amber", "active sessions", active_count, "#active-sessions")
    if cache_age > 3600:
        add("amber", "cache age", f"{cache_age//60}m", "")
    state = "green"
    title = "All clear"
    if any(i["level"] == "red" for i in items):
        state = "red"; title = "Action needed"
    elif items:
        state = "amber"; title = "Attention"
    subtitle_parts = []
    if archive.get("date"):
        subtitle_parts.append(f"latest snapshot {archive.get('date')}")
    if cache.get("layer"):
        subtitle_parts.append(f"cache {cache.get('layer')} age {cache_age}s")
    if not subtitle_parts:
        subtitle_parts.append("live operational summary")
    return {"state": state, "title": title, "items": items, "subtitle": " · ".join(subtitle_parts)}

def compact_snapshot(data: dict[str, Any]) -> dict[str, Any]:
    generated_at = data.get("generated_at") or datetime.now().astimezone().isoformat(timespec="seconds")
    day = str(generated_at)[:10]
    totals = data.get("totals", {}) or {}
    cron_summary = ((data.get("cron_health") or {}).get("summary") or {})
    anomalies = data.get("anomalies") or []
    return {
        "date": day,
        "generated_at": generated_at,
        "period_days": data.get("days"),
        "source": data.get("source"),
        "model": data.get("model"),
        "totals": {k: totals.get(k) for k in ["sessions", "messages", "tool_calls", "input_tokens", "output_tokens", "cache_read_tokens", "cache_write_tokens", "reasoning_tokens", "total_tokens", "cost_usd", "active_time_seconds"]},
        "platforms": data.get("platforms", [])[:20],
        "models": data.get("models", [])[:20],
        "top_tools": data.get("top_tools", [])[:20],
        "top_skills": data.get("top_skills", [])[:20],
        "cron_health": {"summary": cron_summary},
        "anomaly_counts": dict(Counter(a.get("reason") or "unknown" for a in anomalies)),
        "active_sessions": len(data.get("active_sessions") or []),
    }


def write_daily_snapshot(data: dict[str, Any]) -> Path | None:
    # Archive only unfiltered all-source/all-model daily dashboard snapshots.
    if data.get("source") or data.get("model") or int(data.get("days") or 0) != 1:
        return None
    snap = compact_snapshot(data)
    ARCHIVE_DIR.mkdir(parents=True, exist_ok=True)
    path = ARCHIVE_DIR / f"{snap['date']}.json"
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(json.dumps(snap, indent=2, ensure_ascii=False), encoding="utf-8")
    os.replace(tmp, path)
    return path


def load_snapshot_history(limit: int = 60) -> list[dict[str, Any]]:
    if not ARCHIVE_DIR.exists():
        return []
    rows: list[dict[str, Any]] = []
    for path in sorted(ARCHIVE_DIR.glob("*.json"), reverse=True)[:limit]:
        try:
            snap = json.loads(path.read_text())
            totals = snap.get("totals", {}) or {}
            cron = ((snap.get("cron_health") or {}).get("summary") or {})
            rows.append({
                "date": snap.get("date") or path.stem,
                "generated_at": snap.get("generated_at"),
                "sessions": totals.get("sessions") or 0,
                "messages": totals.get("messages") or 0,
                "tool_calls": totals.get("tool_calls") or 0,
                "total_tokens": totals.get("total_tokens") or 0,
                "cost_usd": totals.get("cost_usd") or 0,
                "active_sessions": snap.get("active_sessions") or 0,
                "anomalies": sum((snap.get("anomaly_counts") or {}).values()),
                "cron_overdue": cron.get("overdue") if cron.get("overdue") is not None else (cron.get("stale_enabled_48h") or 0),
                "cron_stale": cron.get("overdue") if cron.get("overdue") is not None else (cron.get("stale_enabled_48h") or 0),
                "cron_delivery_errors": cron.get("delivery_errors") or 0,
                "file": path.name,
            })
        except Exception:
            continue
    return rows

def cache_key(days: int, source: str | None, model: str | None) -> str:
    payload = json.dumps({"days": days, "source": source, "model": model}, sort_keys=True, separators=(",", ":"))
    return hashlib.sha256(payload.encode()).hexdigest()[:24]


def cache_path(days: int, source: str | None, model: str | None) -> Path:
    src = source or "all"
    mdl = model or "all"
    safe = "".join(c if c.isalnum() or c in "._-" else "_" for c in f"d{days}_s{src}_m{mdl}")
    return CACHE_DIR / f"{safe}_{cache_key(days, source, model)}.json"


def read_disk_cache(days: int, source: str | None, model: str | None, max_age: float | None = None) -> dict[str, Any] | None:
    path = cache_path(days, source, model)
    if not path.exists():
        return None
    try:
        age = time.time() - path.stat().st_mtime
        if max_age is not None and age > max_age:
            return None
        data = json.loads(path.read_text())
        # Disk cache files can outlive code changes that add new derived fields.
        # Keep the persistent-speed benefit, but graft cheap live-only metadata that
        # does not require scanning state.db.
        data["archive_history"] = load_snapshot_history(90)
        data["archive_trends"] = build_archive_trends(data["archive_history"])
        data["cache"] = {"layer": "disk", "age_seconds": int(max(0, age)), "path": str(path)}
        data["attention"] = build_attention(data)
        return data
    except Exception:
        return None


def write_disk_cache(days: int, source: str | None, model: str | None, data: dict[str, Any]) -> None:
    CACHE_DIR.mkdir(parents=True, exist_ok=True)
    path = cache_path(days, source, model)
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
    os.replace(tmp, path)


def warm_disk_cache(db: Path, days: int, source: str | None, model: str | None = None, force: bool = False) -> dict[str, Any]:
    if not force:
        cached = read_disk_cache(days, source, model, DISK_CACHE_TTL_SECONDS)
        if cached is not None:
            return cached
    data = load_data(db, days, source, model)
    write_disk_cache(days, source, model, data)
    return data

@closes_read_connections
def load_data(db: Path, days: int, source: str | None, model: str | None = None) -> dict[str, Any]:
    readiness = summary_readiness(db)
    period_end_ts = float(
        readiness["source_observed_at"] or readiness["summary_generated_at"] or 0
    )
    cutoff = period_end_ts - (days * 86400)
    prev_cutoff = cutoff - (days * 86400)
    where = "started_at >= ?"
    params: list[Any] = [cutoff]
    filter_clauses: list[str] = []
    filter_params: list[Any] = []
    if source:
        filter_clauses.append("source = ?")
        filter_params.append(source)
    if model:
        if model == "unknown":
            filter_clauses.append("(model IS NULL OR model = '')")
        else:
            filter_clauses.append("model = ?")
            filter_params.append(model)
    if filter_clauses:
        where += " AND " + " AND ".join(filter_clauses)
        params.extend(filter_params)

    con = open_read_connection(db)

    session_cols = """
        id, source, user_id, model, parent_session_id, started_at, ended_at, end_reason,
        message_count, tool_call_count, input_tokens, output_tokens, title,
        cache_read_tokens, cache_write_tokens, reasoning_tokens, billing_provider,
        billing_mode, estimated_cost_usd, actual_cost_usd, cost_status, cost_source,
        api_call_count, cwd, git_branch, git_repo_root, session_key, chat_id, chat_type, thread_id
    """
    sessions = [dict(r) for r in con.execute(f"SELECT {session_cols} FROM summary_sessions WHERE {where} ORDER BY started_at", params).fetchall()]
    prev_where = "started_at >= ? AND started_at < ?"
    prev_params: list[Any] = [prev_cutoff, cutoff]
    if filter_clauses:
        prev_where += " AND " + " AND ".join(filter_clauses)
        prev_params.extend(filter_params)
    prev_sessions = [dict(r) for r in con.execute(f"SELECT {session_cols} FROM summary_sessions WHERE {prev_where} ORDER BY started_at", prev_params).fetchall()]
    available_sources = sorted({r["source"] or "unknown" for r in con.execute("SELECT DISTINCT source FROM summary_sessions ORDER BY source").fetchall()})
    available_models = sorted({r["model"] or "unknown" for r in con.execute("SELECT DISTINCT model FROM summary_sessions ORDER BY model").fetchall()})

    # Message, tool, and skill aggregates are materialized by the bounded producer.
    # The dashboard never reads raw content, reasoning, API content, or tool arguments.
    msg_where = where
    msg_params = list(params)
    role_counts = {r["role"]: r["n"] for r in con.execute(f"""
        SELECT r.role AS role, SUM(r.count) AS n
        FROM summary_sessions s JOIN summary_session_roles r ON r.session_id = s.id
        WHERE {msg_where}
        GROUP BY r.role
    """, tuple(msg_params)).fetchall()}
    tool_name_rows = query_rows(con, f"""
        SELECT t.tool_name AS name, SUM(t.calls) AS calls
        FROM summary_sessions s JOIN summary_session_tools t ON t.session_id = s.id
        WHERE {msg_where}
        GROUP BY t.tool_name
        ORDER BY calls DESC
        LIMIT 20
    """, tuple(msg_params))
    skill_messages = query_rows(con, f"""
        WITH relevant AS (
            SELECT k.skill_name AS name, k.loads, k.edits, s.started_at,
                   k.first_message_id, k.first_call_index
            FROM summary_sessions s JOIN summary_session_skills k ON k.session_id = s.id
            WHERE {msg_where}
        ), ranked AS (
            SELECT *, ROW_NUMBER() OVER (
                PARTITION BY name
                ORDER BY started_at DESC, first_message_id, first_call_index
            ) AS first_rank
            FROM relevant
        )
        SELECT name, SUM(loads) AS loads, SUM(edits) AS edits,
               MAX(CASE WHEN first_rank = 1 THEN started_at END) AS first_started_at,
               MAX(CASE WHEN first_rank = 1 THEN first_message_id END) AS first_message_id,
               MAX(CASE WHEN first_rank = 1 THEN first_call_index END) AS first_call_index
        FROM ranked
        GROUP BY name
        ORDER BY first_started_at DESC, first_message_id, first_call_index
    """, tuple(msg_params))

    # Active-session durations are part of the committed summary view. Keep them
    # pinned to the producer's observation clock rather than request time.
    now_ts = datetime.fromtimestamp(period_end_ts, timezone.utc).timestamp()
    totals = Counter()
    models = defaultdict(lambda: Counter())
    platforms = defaultdict(lambda: Counter())
    daily = defaultdict(lambda: Counter())
    hours = Counter()
    weekdays = Counter()
    cost_status = Counter()
    recent_sessions = []

    longest = None
    most_messages = None
    most_tokens = None
    most_tools = None

    for s in sessions:
        toks = total_tokens(s)
        dur = max(0.0, float((s.get("ended_at") or now_ts) - (s.get("started_at") or now_ts)))
        cost = s.get("actual_cost_usd") if s.get("actual_cost_usd") is not None else s.get("estimated_cost_usd")
        cost = float(cost or 0)
        totals.update({
            "sessions": 1,
            "messages": int(s.get("message_count") or 0),
            "tool_calls": int(s.get("tool_call_count") or 0),
            "input_tokens": int(s.get("input_tokens") or 0),
            "output_tokens": int(s.get("output_tokens") or 0),
            "cache_read_tokens": int(s.get("cache_read_tokens") or 0),
            "cache_write_tokens": int(s.get("cache_write_tokens") or 0),
            "reasoning_tokens": int(s.get("reasoning_tokens") or 0),
            "total_tokens": toks,
        })
        totals["cost_usd_x10000"] += int(cost * 10000)
        totals["active_seconds"] += int(dur)
        model_name = safe_model(s.get("model"))
        src = s.get("source") or "unknown"
        day = day_key(s.get("started_at"))
        hr = hour_key(s.get("started_at"))
        wd = weekday_key(s.get("started_at"))
        for bucket in [models[model_name], platforms[src], daily[day]]:
            bucket["sessions"] += 1
            bucket["messages"] += int(s.get("message_count") or 0)
            bucket["tool_calls"] += int(s.get("tool_call_count") or 0)
            bucket["tokens"] += toks
            bucket["cost_usd_x10000"] += int(cost * 10000)
        hours[hr] += 1
        weekdays[wd] += 1
        cost_status[s.get("cost_status") or "unknown"] += 1
        packed = {**s, "total_tokens": toks, "duration_seconds": dur, "cost_usd": cost, "started_iso": iso(s.get("started_at")), "ended_iso": iso(s.get("ended_at"))}
        if longest is None or dur > longest["duration_seconds"]: longest = packed
        if most_messages is None or int(s.get("message_count") or 0) > int(most_messages.get("message_count") or 0): most_messages = packed
        if most_tokens is None or toks > most_tokens["total_tokens"]: most_tokens = packed

def latest_output_for_job(job_id: str) -> dict[str, Any]:
    out_dir = HOME / ".hermes" / "cron" / "output" / job_id
    if not out_dir.exists():
        return {"count": 0, "files": []}
    files = [x for x in out_dir.iterdir() if x.is_file()]
    if not files:
        return {"count": 0, "files": []}
    rows = []
    for f in sorted(files, key=lambda x: x.stat().st_mtime, reverse=True)[:12]:
        st = f.stat()
        rows.append({"name": f.name, "mtime": datetime.fromtimestamp(st.st_mtime).astimezone().isoformat(timespec="seconds"), "bytes": st.st_size})
    return {"count": len(files), "latest": rows[0]["name"], "latest_mtime": rows[0]["mtime"], "latest_bytes": rows[0]["bytes"], "files": rows}

        cron_sessions.append({"id": cs.get("id"), "title": cs.get("title") or cs.get("id"), "model": safe_model(cs.get("model")), "started_iso": iso(cs.get("started_at")), "ended_iso": iso(cs.get("ended_at")), "duration_seconds": max(0.0,end-start), "messages": int(cs.get("message_count") or 0), "tool_calls": int(cs.get("tool_call_count") or 0), "total_tokens": total_tokens(cs), "cost_usd": float(cost or 0), "active": cs.get("ended_at") is None})
    return {"job": safe, "sessions": cron_sessions, "privacy": "metadata-only: request bodies, output bodies, script source, and transcripts are intentionally omitted"}

@closes_read_connections
def load_session_detail(db: Path, session_id: str) -> dict[str, Any]:
    con = open_read_connection(db)
    session_cols = """
        id, source, user_id, model, parent_session_id, started_at, ended_at, end_reason,
        message_count, tool_call_count, input_tokens, output_tokens, title,
        cache_read_tokens, cache_write_tokens, reasoning_tokens, billing_provider,
        billing_mode, estimated_cost_usd, actual_cost_usd, cost_status, cost_source,
        api_call_count, cwd, git_branch, git_repo_root, session_key, chat_id, chat_type, thread_id,
        display_name, archived, rewind_count
    """
    row = con.execute(f"SELECT {session_cols} FROM summary_sessions WHERE id = ?", (session_id,)).fetchone()
    if not row:
        raise KeyError("session not found")
    sess = dict(row)
    readiness = summary_readiness(db)
    observed_ts = float(readiness["source_observed_at"] or readiness["summary_generated_at"] or 0)
    now_ts = datetime.fromtimestamp(observed_ts, timezone.utc).timestamp()
    toks = total_tokens(sess)
    start = float(sess.get("started_at") or now_ts)
    end = float(sess.get("ended_at") or now_ts)
    cost = sess.get("actual_cost_usd") if sess.get("actual_cost_usd") is not None else sess.get("estimated_cost_usd")
    sess.update({
        "total_tokens": toks,
        "cost_usd": float(cost or 0),
        "duration_seconds": max(0.0, end - start),
        "started_iso": iso(sess.get("started_at")),
        "ended_iso": iso(sess.get("ended_at")),
        "active": sess.get("ended_at") is None,
    })
    role_counts = {r["role"]: r["n"] for r in con.execute("SELECT role, count AS n FROM summary_session_roles WHERE session_id = ?", (session_id,)).fetchall()}
    tool_rows = [dict(r) for r in con.execute("""
        SELECT tool_name AS name, calls, first_ts, last_ts
        FROM summary_session_tools
        WHERE session_id = ?
        ORDER BY calls DESC, name
        LIMIT 80
    """, (session_id,)).fetchall()]
    for r in tool_rows:
        r["first_iso"] = iso(r.get("first_ts")); r["last_iso"] = iso(r.get("last_ts"))
    timeline = [dict(r) for r in con.execute("""
        SELECT id, role, tool_name, timestamp, token_count, finish_reason, active, compacted,
               content_chars, tool_calls_chars
        FROM summary_message_metadata
        WHERE session_id = ?
        ORDER BY timestamp
        LIMIT 300
    """, (session_id,)).fetchall()]
    for r in timeline:
        r["timestamp_iso"] = iso(r.get("timestamp"))
    skill_rows = [dict(r) for r in con.execute(
        "SELECT skill_name AS name, loads, edits FROM summary_session_skills WHERE session_id = ? ORDER BY loads DESC, first_message_id, first_call_index LIMIT 80",
        (session_id,),
    ).fetchall()]
    return {
        "generated_at": datetime.now(timezone.utc).astimezone().isoformat(timespec="seconds"),
        "session": sess,
        "role_counts": role_counts,
        "tools": tool_rows,
        "skills": skill_rows,
        "timeline": timeline,
        "privacy": "metadata-only: message content, reasoning text, raw tool arguments, and raw tool output are intentionally omitted",
    }

def js_data(data: dict[str, Any]) -> str:
    return json.dumps(data, ensure_ascii=False, separators=(",", ":"))


def render(data: dict[str, Any]) -> str:
    t = data["totals"]
    avg_session = (t.get("active_seconds", 0) / t.get("sessions", 1)) if t.get("sessions") else 0
    avg_msgs = (t.get("messages", 0) / t.get("sessions", 1)) if t.get("sessions") else 0
    max_daily_tokens = max([d["tokens"] for d in data["daily"]] or [1])
    max_hour = max([h["sessions"] for h in data["hours"]] or [1])
    max_weekday = max([w["sessions"] for w in data["weekdays"]] or [1])

    def kpi(label, value, sub="", cls=""):
        return f'<article class="kpi {cls}"><span>{escape(label)}</span><strong>{escape(value)}</strong><em>{escape(sub)}</em></article>'

    def bar_rows(rows, name_key, val_key, label_fmt=lambda r: ""):
        out = []
        maxv = max([r[val_key] for r in rows] or [1])
        for r in rows:
            pct = 0 if maxv == 0 else max(2, r[val_key] / maxv * 100)
            out.append(f"""
            <div class="bar-row">
              <div class="bar-label" title="{escape(str(r[name_key]))}">{escape(str(r[name_key]))}</div>
              <div class="bar-track"><div class="bar-fill" style="width:{pct:.2f}%"></div></div>
              <div class="bar-value">{escape(label_fmt(r) or compact_num(r[val_key]))}</div>
            </div>""")
        return "\n".join(out) or '<p class="muted">No data.</p>'

    daily_bars = "".join(f"""
      <div class="daily-bar" title="{escape(d['date'])}: {fmt_int(d['tokens'])} tokens">
        <div class="daily-fill" style="height:{(d['tokens']/max_daily_tokens*100 if max_daily_tokens else 0):.2f}%"></div>
        <span>{escape(d['date'][5:] if len(d['date']) > 5 else d['date'])}</span>
      </div>""" for d in data["daily"][-30:])

    hour_bars = "".join(f"""
      <div class="hour" title="{h['hour']:02d}:00 - {h['sessions']} sessions">
        <div style="height:{(h['sessions']/max_hour*100 if max_hour else 0):.2f}%"></div><span>{h['hour']}</span>
      </div>""" for h in data["hours"])

    weekday_rows = "".join(f"""
      <div class="weekday"><span>{w['day']}</span><div><b style="width:{(w['sessions']/max_weekday*100 if max_weekday else 0):.2f}%"></b></div><em>{w['sessions']}</em></div>""" for w in data["weekdays"])

    def session_card(title, s):
        if not s:
            return ""
        return f"""
        <article class="session-card">
          <span>{escape(title)}</span>
          <strong title="{escape(s.get('id',''))}">{escape((s.get('title') or s.get('id') or 'unknown')[:56])}</strong>

            <dt>Started</dt><dd>{escape(s.get('started_iso') or '')}</dd>

        </article>"""

    recent_rows = "".join(f"""
      <tr>
        <td><code title="{escape(s.get('id') or '')}">{escape((s.get('id') or '')[:22])}</code></td>
        <td>{escape(s.get('source') or 'unknown')}</td>

        <div class="session-meta"><em>{escape(s.get('model') or 'unknown')}</em><em>{escape(duration(s.get('duration_seconds')))}</em><em>{fmt_int(s.get('total_tokens'))} tok</em></div>

        </dl>
      </article>""" for s in data.get("active_sessions", [])) or '<p class="muted">No currently active sessions in this period.</p>'

    def option_rows(values, selected):
        rows = [f'<option value="" {"selected" if not selected else ""}>all</option>']
        for v in values:
            val = str(v or 'unknown')
            rows.append(f'<option value="{escape(val)}" {"selected" if selected == val else ""}>{escape(val)}</option>')

def serve(db: Path, host: str, port: int, default_days: int, default_source: str | None) -> None:
    cache: dict[tuple[int, str | None, str | None], dict[str, Any]] = {}
    cache_lock = threading.Lock()
    refresh_ttl = 60.0
    service_started_ts = time.time()
    prewarm_state: dict[str, Any] = {"state": "starting", "completed": 0, "errors": 0, "total": len(PREWARM_DAYS) * len(PREWARM_SOURCES), "last_started_at": datetime.now().astimezone().isoformat(timespec="seconds"), "last_finished_at": None, "last_error": None}
    cache_retention_days = int(os.environ.get("HERMES_INSIGHTS_CACHE_RETENTION_DAYS", "14"))
    archive_retention_days = int(os.environ.get("HERMES_INSIGHTS_ARCHIVE_RETENTION_DAYS", "365"))

    def retention_status() -> dict[str, Any]:
        now = time.time()
        cache_files = list(CACHE_DIR.glob("*.json")) if CACHE_DIR.exists() else []
        archive_files = list(ARCHIVE_DIR.glob("*.json")) if ARCHIVE_DIR.exists() else []
        def older(files: list[Path], days: int) -> list[Path]:
            cutoff = now - days * 86400
            return [f for f in files if f.stat().st_mtime < cutoff]
        old_cache = older(cache_files, cache_retention_days)
        old_archive = older(archive_files, archive_retention_days)
        return {
            "cache_policy_days": cache_retention_days,
            "archive_policy_days": archive_retention_days,
            "old_cache_files": len(old_cache),
            "old_archive_files": len(old_archive),
            "cache_files": len(cache_files),
            "archive_files": len(archive_files),
            "cleanup_script": "hermes_insights_daily_snapshot.sh",
        }

    def self_health() -> dict[str, Any]:
        now = time.time()
        disk_files = list(CACHE_DIR.glob('*.json')) if CACHE_DIR.exists() else []
        archive_files = sorted(ARCHIVE_DIR.glob('*.json')) if ARCHIVE_DIR.exists() else []
        db_stat = db.stat() if db.exists() else None
        with cache_lock:
            entries = len(cache)
            refreshing = sum(1 for e in cache.values() if e.get("refreshing"))
            errors = [str(e.get("error")) for e in cache.values() if e.get("error")]
            prewarm = dict(prewarm_state)
        return {
            "ok": True,
            "service_started_at": datetime.fromtimestamp(service_started_ts).astimezone().isoformat(timespec="seconds"),
            "uptime_seconds": int(now - service_started_ts),
            "cache_entries": entries,
            "refreshing_entries": refreshing,
            "disk_cache_files": len(disk_files),
            "disk_cache_bytes": sum(f.stat().st_size for f in disk_files) if disk_files else 0,
            "archive_files": len(archive_files),
            "latest_archive": archive_files[-1].name if archive_files else None,
            "db_mtime": datetime.fromtimestamp(db_stat.st_mtime).astimezone().isoformat(timespec="seconds") if db_stat else None,
            "db_size_bytes": db_stat.st_size if db_stat else None,
            "prewarm": prewarm,
            "retention": retention_status(),
            "last_refresh_errors": errors[:5],
        }

    def make_entry(data: dict[str, Any], layer: str, ts: float | None = None) -> dict[str, Any]:
        now = time.time()
        ts = ts or now
        payload = dict(data)
        payload["cache"] = {"layer": layer, "age_seconds": int(max(0, now - ts))}
        return {"data": payload, "json": json.dumps(payload, indent=2, ensure_ascii=False), "ts": ts, "refreshing": False, "error": None, "layer": layer}

    def refresh_key(key: tuple[int, str | None, str | None]) -> None:
        days, source, model = key
        try:
            data = load_data(db, days, source, model)
            write_daily_snapshot(data)
            write_disk_cache(days, source, model, data)
            entry = make_entry(data, "fresh", time.time())
            with cache_lock:
                cache[key] = entry
        except Exception as e:
            with cache_lock:
                entry = cache.setdefault(key, {})
                entry["refreshing"] = False
                entry["error"] = str(e)

    def prewarm_common() -> None:
        with cache_lock:
            prewarm_state.update({"state": "warming", "last_started_at": datetime.now().astimezone().isoformat(timespec="seconds"), "last_finished_at": None})
        for d in PREWARM_DAYS:
            for src in PREWARM_SOURCES:
                try:
                    key = (d, src, None)
                    cached = read_disk_cache(d, src, None, DISK_CACHE_TTL_SECONDS)
                    if cached is not None:
                        with cache_lock:
                            cache[key] = make_entry(cached, "disk", time.time() - int(cached.get("cache", {}).get("age_seconds", 0)))
                            prewarm_state["completed"] = int(prewarm_state.get("completed") or 0) + 1
                        continue
                    refresh_key(key)
                    with cache_lock:
                        prewarm_state["completed"] = int(prewarm_state.get("completed") or 0) + 1
                except Exception as e:
                    with cache_lock:
                        prewarm_state["errors"] = int(prewarm_state.get("errors") or 0) + 1
                        prewarm_state["last_error"] = str(e)
                    print(f"prewarm failed days={d} source={src}: {e}")
        with cache_lock:
            prewarm_state["state"] = "idle"
            prewarm_state["last_finished_at"] = datetime.now().astimezone().isoformat(timespec="seconds")

    def get_entry(days: int, source: str | None, model: str | None) -> dict[str, Any]:
        key = (days, source, model)
        now = time.time()
        with cache_lock:
            entry = cache.get(key)
            if entry and now - float(entry.get("ts") or 0) <= refresh_ttl:
                return entry
            if entry and not entry.get("refreshing"):
                entry["refreshing"] = True
                threading.Thread(target=refresh_key, args=(key,), daemon=True).start()
                return entry
            if entry and entry.get("refreshing"):
                return entry
        disk_data = read_disk_cache(days, source, model, None)
        if disk_data is not None:
            path = cache_path(days, source, model)
            ts = path.stat().st_mtime if path.exists() else now
            entry = make_entry(disk_data, "disk", ts)
            with cache_lock:
                cache[key] = entry
                if now - ts > DISK_CACHE_TTL_SECONDS and not entry.get("refreshing"):
                    entry["refreshing"] = True
                    threading.Thread(target=refresh_key, args=(key,), daemon=True).start()
            return entry
        with cache_lock:
            cache[key] = {"refreshing": True, "ts": 0}
        refresh_key(key)
        with cache_lock:
            return cache[key]

    threading.Thread(target=prewarm_common, daemon=True).start()

    class Handler(BaseHTTPRequestHandler):
        server_version = "HermesInsightsDashboard/1.0"

        def log_message(self, fmt: str, *args: Any) -> None:
            print(f"{self.address_string()} - {fmt % args}")

        def _params(self) -> tuple[int, str | None, str | None]:
            qs = parse_qs(urlparse(self.path).query)
            try:
                days = int((qs.get("days") or [default_days])[0])
            except Exception:
                days = default_days
            days = max(1, min(days, 365))
            source = (qs.get("source") or [default_source])[0]
            source = source or None
            model = (qs.get("model") or [None])[0]
            model = model or None
            return days, source, model

        def _send(self, status: int, body: bytes, ctype: str, age: int | None = None, headers: dict[str, str] | None = None) -> None:
            self.send_response(status)
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.send_header("Cache-Control", "no-store, no-cache, must-revalidate, max-age=0")
            self.send_header("Pragma", "no-cache")
            if age is not None:
                self.send_header("X-Hermes-Cache-Age", str(age))
            for hk, hv in (headers or {}).items():
                self.send_header(hk, hv)
            self.end_headers()
            self.wfile.write(body)

        def _json_response(self, payload: Any, filename: str | None = None, age: int | None = None) -> None:
            body = json.dumps(payload, indent=2, ensure_ascii=False).encode()
            headers = {}
            if filename:
                safe = re.sub(r"[^A-Za-z0-9_.-]+", "-", filename).strip("-") or "hermes-insights.json"
                headers["Content-Disposition"] = f'attachment; filename="{safe}"'
            self._send(200, body, "application/json; charset=utf-8", age=age, headers=headers)

        def do_GET(self) -> None:
            path = urlparse(self.path).path
            qs = parse_qs(urlparse(self.path).query)
            download = (qs.get("download") or [""])[0] in ("1", "true", "yes")
            if path == "/archive-history.json":
                self._json_response({"ok": True, "generated_at": datetime.now().astimezone().isoformat(timespec="seconds"), "archive_history": load_snapshot_history(365), "archive_trends": build_archive_trends(load_snapshot_history(365))}, filename="hermes-insights-archive-history.json")
                return
            if path == "/export/health.json":
                h = self_health(); data = load_data(db, 1, None, None); h.update({"generated_at": data["generated_at"], "sessions": data["totals"].get("sessions", 0)})
                self._json_response(h, filename="hermes-insights-health.json")
                return
            if path == "/export/archive-history.json":
                self._json_response({"ok": True, "generated_at": datetime.now().astimezone().isoformat(timespec="seconds"), "archive_history": load_snapshot_history(365), "archive_trends": build_archive_trends(load_snapshot_history(365))}, filename="hermes-insights-archive-history.json")
                return
            if path in ("/export/current.json", "/export/cron-health.json", "/export/attention.json"):
                days, source, model = self._params()
                entry = get_entry(days, source, model)
                if "data" not in entry:
                    err = entry.get("error") or "cache warming failed"
                    self._send(500, f"dashboard error: {escape(str(err))}".encode(), "text/plain; charset=utf-8")
                    return
                payload = dict(entry["data"]); payload["dashboard_self_health"] = self_health()
                suffix = f"{days}d" + (f"-{source}" if source else "-all") + (f"-{model}" if model else "")
                if path == "/export/current.json":
                    self._json_response(payload, filename=f"hermes-insights-current-{suffix}.json", age=max(0, int(time.time() - float(entry.get("ts") or 0))))
                elif path == "/export/cron-health.json":
                    self._json_response({"ok": True, "generated_at": payload.get("generated_at"), "days": days, "source": source, "model": model, "cron_health": payload.get("cron_health")}, filename=f"hermes-insights-cron-health-{suffix}.json")
                else:
                    self._json_response({"ok": True, "generated_at": payload.get("generated_at"), "days": days, "source": source, "model": model, "attention": payload.get("attention")}, filename=f"hermes-insights-attention-{suffix}.json")
                return
            if path in ("/health", "/readyz"):
                try:
                    h = self_health()
                    readiness = summary_readiness(db)
                    data = load_data(db, 1, None, None)
                    h.update(readiness)
                    h.update({"ok": True if path == "/health" else readiness["ready"], "generated_at": data["generated_at"], "sessions": data["totals"].get("sessions", 0)})
                    body = json.dumps(h, indent=2, ensure_ascii=False).encode()
                    status = 200 if path == "/health" or readiness["ready"] else 503
                    self._send(status, body, "application/json; charset=utf-8")
                except Exception as e:
                    payload = {"ok": path == "/health", "ready": False, "data_status": "error", "error": str(e)}
                    self._send(200 if path == "/health" else 503, json.dumps(payload).encode(), "application/json; charset=utf-8")
                return
            if path.startswith("/cron-data/"):
                job_id = path.split("/cron-data/", 1)[1]
                try:
                    from urllib.parse import unquote
                    data = load_cron_detail(db, unquote(job_id))
                    self._send(200, json.dumps(data, indent=2, ensure_ascii=False).encode(), "application/json; charset=utf-8")
                except KeyError:
                    self._send(404, json.dumps({"ok": False, "error": "cron job not found"}).encode(), "application/json; charset=utf-8")
                except Exception as e:
                    self._send(500, json.dumps({"ok": False, "error": str(e)}).encode(), "application/json; charset=utf-8")
                return
            if path.startswith("/cron/"):
                from urllib.parse import unquote
                job_id = unquote(path.split("/cron/", 1)[1])
                if not job_id:
                    self._send(404, b"not found", "text/plain; charset=utf-8")
                    return
                self._send(200, render_cron_shell(job_id).encode(), "text/html; charset=utf-8", age=0)
                return
            if path.startswith("/session-data/"):
                session_id = path.split("/session-data/", 1)[1]
                try:
                    from urllib.parse import unquote
                    data = load_session_detail(db, unquote(session_id))
                    self._send(200, json.dumps(data, indent=2, ensure_ascii=False).encode(), "application/json; charset=utf-8")
                except KeyError:
                    self._send(404, json.dumps({"ok": False, "error": "session not found"}).encode(), "application/json; charset=utf-8")
                except Exception as e:
                    self._send(500, json.dumps({"ok": False, "error": str(e)}).encode(), "application/json; charset=utf-8")
                return
            if path.startswith("/session/"):
                from urllib.parse import unquote
                session_id = unquote(path.split("/session/", 1)[1])
                if not session_id:
                    self._send(404, b"not found", "text/plain; charset=utf-8")
                    return
                self._send(200, render_session_shell(session_id).encode(), "text/html; charset=utf-8", age=0)
                return
            if path not in ("/", "/index.html", "/index.json", "/data.json"):
                self._send(404, b"not found", "text/plain; charset=utf-8")
                return
            if path in ("/", "/index.html"):
                self._send(200, render_shell().encode(), "text/html; charset=utf-8", age=0)
                return
            try:
                days, source, model = self._params()
                entry = get_entry(days, source, model)
                if "data" not in entry:
                    err = entry.get("error") or "cache warming failed"
                    self._send(500, f"dashboard error: {escape(str(err))}".encode(), "text/plain; charset=utf-8")
                    return
                age = max(0, int(time.time() - float(entry.get("ts") or 0)))
                payload = dict(entry["data"])
                payload["dashboard_self_health"] = self_health()
                filename = None
                if download:
                    suffix = f"{days}d" + (f"-{source}" if source else "-all") + (f"-{model}" if model else "")
                    filename = f"hermes-insights-current-{suffix}.json"
                self._json_response(payload, filename=filename, age=age)
            except Exception as e:
                self._send(500, f"dashboard error: {escape(str(e))}".encode(), "text/plain; charset=utf-8")

    httpd = ThreadingHTTPServer((host, port), Handler)
    print(f"Serving Hermes Insights dashboard on http://{host}:{port} from {db}")
    httpd.serve_forever()


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--db", type=Path, default=DEFAULT_DB)
    ap.add_argument("--out", type=Path, default=DEFAULT_OUT)
    ap.add_argument("--days", type=int, default=7)
    ap.add_argument("--source")
    ap.add_argument("--model")
    ap.add_argument("--serve", action="store_true", help="Run an HTTP server that regenerates data on every request")
    ap.add_argument("--host", default="127.0.0.1")
    ap.add_argument("--port", type=int, default=18766)
    ap.add_argument("--require-ready", action="store_true", help="Refuse output unless the compact summary is complete and fresh")
    ap.add_argument("--max-age-seconds", type=float, default=1800)
    args = ap.parse_args()
    if args.serve:
        serve(args.db, args.host, args.port, args.days, args.source)
        return
    if args.require_ready:
        readiness = summary_readiness(args.db, max_age_seconds=args.max_age_seconds)
        if not readiness["ready"]:
            raise SystemExit("summary is not ready: " + readiness["data_status"])
    data = load_data(args.db, args.days, args.source, args.model)
    snapshot_path = write_daily_snapshot(data)
    args.out.parent.mkdir(parents=True, exist_ok=True)
    args.out.write_text(render(data), encoding="utf-8")
    json_path = args.out.with_suffix(".json")
    json_path.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
    print(json.dumps({"html": str(args.out), "json": str(json_path), "sessions": data["totals"].get("sessions",0), "tokens": data["totals"].get("total_tokens",0), "snapshot": str(snapshot_path) if snapshot_path else None}, indent=2))

if __name__ == "__main__":
    main()
